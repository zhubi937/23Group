cmake_minimum_required(VERSION 3.16)
project(jtocr LANGUAGES C CXX)
cmake_policy(SET CMP0137 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0137 NEW)
list(APPEND CMAKE_TRY_COMPILE_PLATFORM_VARIABLES OHOS_SDK_NATIVE HMOS_SDK_NATIVE OHOS_ARCH)
set_property(GLOBAL PROPERTY JOB_POOLS jt_compile_pool=4)
set(CMAKE_JOB_POOL_COMPILE jt_compile_pool)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SW_BUILD OFF CACHE BOOL "" FORCE)
set(BUILD_PROG OFF CACHE BOOL "" FORCE)
foreach(codec ZLIB PNG GIF JPEG TIFF WEBP OPENJPEG)
  set(ENABLE_${codec} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_TRAINING_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(OPENMP_BUILD OFF CACHE BOOL "" FORCE)
set(GRAPHICS_DISABLED ON CACHE BOOL "" FORCE)
set(DISABLED_LEGACY_ENGINE ON CACHE BOOL "" FORCE)
set(DISABLE_TIFF ON CACHE BOOL "" FORCE)
set(DISABLE_ARCHIVE ON CACHE BOOL "" FORCE)
set(DISABLE_CURL ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/leptonica-1.85.0 leptonica EXCLUDE_FROM_ALL)
file(GLOB LEPT_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/leptonica-1.85.0/src/*.h")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/leptonica")
file(COPY ${LEPT_HEADERS} DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/leptonica")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/leptonica/src/endianness.h" "${CMAKE_CURRENT_BINARY_DIR}/include/leptonica/endianness.h" COPYONLY)
set(JT_LEPT_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/include")
add_subdirectory(third_party/tesseract-5.5.0 tesseract EXCLUDE_FROM_ALL)
target_include_directories(libtesseract PRIVATE "${JT_LEPT_INCLUDE}/leptonica")
add_library(jtocr SHARED ocr_napi.cpp)
target_include_directories(jtocr PRIVATE "${JT_LEPT_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}/tesseract/include")
target_link_libraries(jtocr PRIVATE libtesseract leptonica libace_napi.z.so libhilog_ndk.z.so)
target_compile_options(jtocr PRIVATE -Wall -Wextra -Werror=return-type)
